cd /scratch/firenze/NN

# Submit fresh strict preflights after synchronizing this suite. They execute
# real UMAP, PCA, and Leiden work and validate the revised public API.
bash benchmark_scripts/fastembedr_jss_review_validation/submit_preflight.sh

# Wait until the two preflight jobs leave squeue, then verify their evidence.
bash benchmark_scripts/fastembedr_jss_review_validation/verify_preflight.sh

# Start the complete campaign. One controller submits bounded waves and
# retries QOS submission-limit errors automatically after 60 seconds.
bash \
  benchmark_scripts/fastembedr_jss_review_validation/submit_complete_campaign.sh

# The launcher prints the unique campaign directory. Monitoring is optional.
squeue -u "$USER"

# Do not run submit_all.sh; its monolithic submission can exceed the QOS limit.

# The controller uses afterany between bounded waves. Partial worker failures
# therefore reach the strict final audit instead of freezing the campaign.

# Focused observed-KNN-recall and backend-quality campaign. This launcher uses
# 11 dataset tasks per backend instead of 132 quality tasks per backend, which
# keeps the submission below the account's queued-job limit. Each dataset task
# continues through all 12 quality configurations and records failed subruns.
bash benchmark_scripts/fastembedr_jss_review_validation/submit_recall_quality_compact.sh

# Optional focused submission for held-out transformation and 20% landmark
# reconstruction. Both families test compact-support t-SNE and fuzzy UMAP on
# all datasets. CUDA is gated by its own strict preflight.
CPU_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cpu.sh)
CUDA_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cuda.sh)
PRECOMPUTE_JOB=$(sbatch --parsable --dependency="afterok:$CPU_PREFLIGHT_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_precompute_cpu12.sh)
sbatch --dependency="afterok:$PRECOMPUTE_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_transform_cpu4.sh
sbatch --dependency="afterok:$PRECOMPUTE_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_landmark_reconstruction_cpu4.sh
sbatch --dependency="afterok:$PRECOMPUTE_JOB:$CUDA_PREFLIGHT_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_transform_cuda.sh
sbatch --dependency="afterok:$PRECOMPUTE_JOB:$CUDA_PREFLIGHT_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_landmark_reconstruction_cuda.sh

# Optional focused PCA accuracy validation. This is separate from PCA timing
# so the dense reference calculation does not contaminate runtime or memory.
CPU_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cpu.sh)
CUDA_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cuda.sh)
sbatch --dependency="afterok:$CPU_PREFLIGHT_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_pca_accuracy_cpu4.sh
sbatch --dependency="afterok:$CUDA_PREFLIGHT_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_pca_accuracy_cuda.sh

# Optional focused clustering validation. Graph construction runs once per
# dataset. CPU Louvain, Leiden, and Walktrap run before CUDA Louvain and Leiden
# so CUDA membership agreement can use the completed CPU reference files.
CPU_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cpu.sh)
CUDA_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cuda.sh)
GRAPH_JOB=$(sbatch --parsable \
  --dependency="afterok:$CPU_PREFLIGHT_JOB" \

CPU_CLUSTER_JOB=$(sbatch --parsable \
  --dependency="afterok:$GRAPH_JOB" \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_clustering_cpu4.sh)
sbatch --dependency="afterok:$CPU_CLUSTER_JOB:$CUDA_PREFLIGHT_JOB" \

# Do not submit run_backend_quality_cpu4.sh and run_backend_quality_cuda.sh
# together on accounts with a small QOS submission limit. Those diagnostic

# Optional focused submission for the long-run t-SNE correctness issue.
CPU_PREFLIGHT_JOB=$(sbatch --parsable \
  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cpu.sh)

  benchmark_scripts/fastembedr_jss_review_validation/slurm/run_preflight_cuda.sh)
INPUT_JOB=$(sbatch --parsable --dependency="afterok:$CPU_PREFLIGHT_JOB" \

printf 'cpu_preflight=%s cuda_preflight=%s input=%s cpu=%s cpu_final=%s threads=%s python=%s cuda=%s cuda_final=%s timing_cpu=%s timing_python=%s timing_cuda=%s\n' \
